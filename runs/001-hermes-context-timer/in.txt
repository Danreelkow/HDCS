#!/usr/bin/env bash
# sync-hermes-context.sh — one-way host->workspace mirror of hermes-context (MIRROR class: contents + recursive dirs + symlinks; not metadata)
# Laws: A1-A21. Staged copy is content-verified (diff -r) before DST is ever touched (A11/A13).
set -u

PROG="$(basename "$0")"
DEFAULT_SRC="/opt/data/workspace/hermes-context"
DEFAULT_DST="/workspace/hermes-context"
LOG_FILE_DEFAULT="${HOME}/.cache/hermes-context/sync.log"

DRY_RUN=0
VERIFY_ONLY=0
for arg in "$@"; do
  case "$arg" in
    --dry-run) DRY_RUN=1 ;;
    --verify)  VERIFY_ONLY=1 ;;
    *) echo "usage: $PROG [--dry-run] [--verify]" >&2; exit 64 ;;
  esac
done

# --- canonicalize inputs (A17: env override is the mandated contract mechanism) ---
canon_path() { # strip redundant trailing slashes but keep "/", no fs access
  local p="$1"
  while [ "$p" != "/" ] && [ "${p%/}" != "$p" ]; do p="${p%/}"; done
  printf '%s' "$p"
}
SRC="$(canon_path "${HERMES_CONTEXT_SRC:-$DEFAULT_SRC}")"
DST="$(canon_path "${HERMES_CONTEXT_DST:-$DEFAULT_DST}")"

refuse() { echo "refused: A$1: $2" >&2; exit 1; }

# --- A18: degenerate paths (component-split tests, not slash-suffix strings) ---
for P in "$SRC" "$DST"; do
  case "$P" in
    ""|"/"|".") refuse 18 "degenerate path '$P' (empty, / or .)" ;;
  esac
done

# --- A12: realpath-based identity / ancestor / descendant containment ---
R_SRC="$(realpath -m -- "$SRC")" || refuse 12 "cannot resolve SRC '$SRC'"
R_DST="$(realpath -m -- "$DST")" || refuse 12 "cannot resolve DST '$DST'"
if [ "$R_SRC" = "$R_DST" ]; then
  refuse 12 "realpath(SRC) == realpath(DST) ($R_SRC)"
fi
# component-split containment, either direction
path_within() { # $1 = candidate child, $2 = candidate ancestor
  local c="$1" a="$2"
  [ "$c" = "$a" ] && return 0
  case "$c/" in "$a"/*) return 0 ;; esac
  return 1
}
if path_within "$R_DST" "$R_SRC"; then
  refuse 12 "DST resolves inside SRC ($R_DST is within $R_SRC)"
fi
if path_within "$R_SRC" "$R_DST"; then
  refuse 12 "SRC is inside DST ($R_SRC is within $R_DST) — sync would destroy the source"
fi

# --- A14/A15: owned concrete instantiated paths vs SRC/DST, boundary-aware ---
LOG_FILE="$LOG_FILE_DEFAULT"
LOG_DIR="$(canon_path "$(dirname "$LOG_FILE")")"
ENTRYPOINT_DIR="$(cd "$(dirname "$0")" 2>/dev/null && pwd)" || ENTRYPOINT_DIR=""
STAGE_BASE="$LOG_DIR"                 # stage lives beside the log, under script-owned cache dir
STAGE_PATH_STR="$STAGE_BASE/.hc-stage.$$"   # A20: pure string computation, no fs access yet

check_owned() { # $1 = concrete path, $2 = label
  local p="$(realpath -m -- "$1")" label="$2"
  [ -n "$p" ] || return 0
  if [ "$p" = "$R_SRC" ] || [ "$p" = "$R_DST" ]; then
    refuse 14 "$label '$p' collides with SRC or DST"
  fi
  if path_within "$p" "$R_SRC" || path_within "$R_SRC" "$p"; then
    refuse 14 "$label '$p' overlaps SRC boundary ($R_SRC)"
  fi
  if path_within "$p" "$R_DST" || path_within "$R_DST" "$p"; then
    refuse 15 "$label '$p' overlaps DST boundary ($R_DST)"
  fi
}
check_owned "$LOG_DIR"      "log dir"
check_owned "$LOG_FILE"     "log file"
check_owned "$STAGE_PATH_STR" "staging path"
if [ -n "$ENTRYPOINT_DIR" ] && [ "$ENTRYPOINT_DIR" != "/" ]; then
  check_owned "$ENTRYPOINT_DIR" "entrypoint dir"
fi

# --- verify-only mode: read-only A9-class comparison of DST vs SRC ---
verify_mirror() { # $1=reference $2=candidate ; contents, recursive dirs, symlink targets
  diff -r --no-dereference -q -- "$1" "$2" >/dev/null 2>&1
}
if [ "$VERIFY_ONLY" -eq 1 ]; then
  if [ ! -d "$DST" ]; then
    echo "refused: A13: --verify failed: destination '$DST' does not exist" >&2
    exit 1
  fi
  if verify_mirror "$SRC" "$DST"; then
    echo "verify ok: $DST matches $SRC (MIRROR class)"
    exit 0
  else
    echo "verify FAILED: $DST does not match $SRC (A9)" >&2
    exit 1
  fi
fi

# --- A6/A16: dry-run is pure plan on stdout — zero writes, no log, no mkdir, no mktemp ---
if [ "$DRY_RUN" -eq 1 ]; then
  echo "PLAN (dry-run, zero side effects):"
  echo "  source:      $SRC"
  echo "  destination: $DST"
  echo "  mode:        one-way mirror (contents + recursive dirs + symlinks), stale entries deleted at all depths (A5/A7)"
  echo "  strategy:    stage -> content-verify -> replace DST (rsync primary, tar fallback)"
  echo "  log target:  $LOG_FILE (not written in dry-run)"
  exit 0
fi

# --- real run from here: writes permitted (A16) ---
mkdir -p -- "$LOG_DIR" || { echo "refused: A14: cannot create log dir '$LOG_DIR'" >&2; exit 1; }
log() { printf '%s %s: %s\n' "$(date '+%Y-%m-%dT%H:%M:%S%z')" "$PROG" "$*" >> "$LOG_FILE"; }

log "start src=$SRC dst=$DST"

# A20: create stage only after string validation; reserved namespace .hc-stage*
STAGE="$(mktemp -d "$STAGE_BASE/.hc-stage.XXXXXX")" || { log "FAIL: cannot create staging dir"; echo "error: cannot create staging dir" >&2; exit 1; }

cleanup_stage() { rm -rf -- "$STAGE"; }
trap 'cleanup_stage' EXIT

# --- stage copy: rsync primary (A4), tar fallback; contents of SRC into stage, no nesting (A4/A5) ---
if command -v rsync >/dev/null 2>&1; then
  if ! rsync -a --delete --copy-links=no -- "$SRC/" "$STAGE/"; then
    log "FAIL: rsync staging failed"; echo "error: rsync staging failed" >&2; exit 1
  fi
else
  # fallback: tar-pipe contents of SRC into stage
  if ! tar -C "$SRC" -cf - . | tar -C "$STAGE" -xf -; then
    log "FAIL: tar fallback staging failed"; echo "error: tar fallback staging failed" >&2; exit 1
  fi
  # A7/A20: recursive reconcile with prunelist — delete stale entries at all depths
  PRUNELIST="$(mktemp "$STAGE/.prunelist.XXXXXX")" || { log "FAIL: prunelist mktemp failed"; exit 1; }
  SRCLIST="$(mktemp "$STAGE/.prunelist.XXXXXX")" || { rm -f -- "$PRUNELIST"; log "FAIL: srclist mktemp failed"; exit 1; }
  ( cd "$STAGE" && find . -mindepth 1 -print0 ) | sort -z > "$PRUNELIST"
  ( cd "$SRC"   && find . -mindepth 1 -print0 ) | sort -z > "$SRCLIST"
  # entries present in stage but absent in SRC are stale
  comm -23 "$PRUNELIST" "$SRCLIST" | while IFS= read -r -d '' stale; do
    rm -rf -- "$STAGE/$stale"
  done
  rm -f -- "$PRUNELIST" "$SRCLIST"
fi

# --- A9/A13: content-verify stage vs SRC BEFORE touching DST; FAIL nonzero, never warn-exit-0 ---
if ! verify_mirror "$SRC" "$STAGE"; then
  log "FAIL: staged copy verification failed (A9) — DST untouched"
  echo "error: staged copy failed verification against source (A9); destination untouched" >&2
  exit 1
fi
log "staged copy verified against source"

# --- A11: VERIFIED_COPY exists (in stage) — only now may DST be replaced ---
mkdir -p -- "$(dirname -- "$DST")" || { log "FAIL: cannot create DST parent"; echo "error: cannot create DST parent" >&2; exit 1; }
if [ -L "$DST" ]; then
  # A18: DST was a symlink passing guards — replace with the real tree, never follow
  rm -f -- "$DST"
fi
if mv -- "$STAGE" "$DST" 2>/dev/null; then
  trap - EXIT
  log "done: DST replaced atomically via mv"
else
  # cross-fs: converge via rsync then remove stage
  if command -v rsync >/dev/null 2>&1; then
    rsync -a --delete --copy-links=no -- "$STAGE/" "$DST/" || { log "FAIL: cross-fs rsync into DST failed"; echo "error: cross-fs convergence failed" >&2; exit 1; }
    rm -rf -- "$STAGE"; trap - EXIT
    log "done: DST converged via cross-fs rsync"
  else
    log "FAIL: mv failed and no rsync available for cross-fs convergence"
    echo "error: cannot converge DST (mv failed, no rsync)" >&2
    exit 1
  fi
fi
log "end ok"
exit 0

=== README.md ===
# hermes-context sync

One-way mirror of the hermes-context tree from the host mount into the
workspace. Source defaults to `/opt/data/workspace/hermes-context/`,
destination to `/workspace/hermes-context/`.

Mirror class: file contents + recursive directory structure + symlinks.
File metadata (timestamps, ownership, hardlinks) is NOT mirrored. The
destination end state always equals the source end state: stale entries
are deleted recursively at every depth, and file<->dir / symlink type
changes are reconciled.

## Install

```sh
mkdir -p ~/.local/bin ~/.config/systemd/user
cp sync-hermes-context.sh ~/.local/bin/sync-hermes-context.sh
chmod +x ~/.local/bin/sync-hermes-context.sh
cp hermes-context.service hermes-context.timer ~/.config/systemd/user/
systemctl --user daemon-reload
systemctl --user enable --now hermes-context.timer
```

The service runs `/home/<user>/.local/bin/sync-hermes-context.sh`
(ExecStart=%h/.local/bin/sync-hermes-context.sh) as a oneshot user unit —
no root. The script also runs standalone (e.g. from cron) without systemd.

## Environment override (contract mechanism)

```sh
HERMES_CONTEXT_SRC=/path/to/source HERMES_CONTEXT_DST=/path/to/dest \
  ~/.local/bin/sync-hermes-context.sh
```

## Dry-run test (zero side effects)

```sh
HERMES_CONTEXT_SRC=/path/to/source HERMES_CONTEXT_DST=/path/to/dest \
  ~/.local/bin/sync-hermes-context.sh --dry-run
```

`--dry-run` prints the plan to stdout only: no files or directories are
created anywhere, no log entry is written, and an absent destination stays
absent. Confirm the destination is untouched afterwards.

## Verify

```sh
~/.local/bin/sync-hermes-context.sh --verify
```

Read-only comparison of destination vs source (contents, structure,
symlink targets). Exits nonzero on any mismatch, including an absent
destination.

## Sync strategy

- Primary: the source contents are copied with `rsync -a --delete` into a
  script-owned staging directory, then content-verified against the source
  (recursive diff of files, dirs, symlink targets).
- Fallback (no rsync): a tar pipe copies the source contents into staging,
  followed by a recursive reconcile (prunelist) that deletes stale entries
  at all depths.
- Only after the staged copy verifies is the destination replaced. A
  verification failure exits nonzero and leaves the destination untouched.
- Both paths converge to the exact same recursive mirror end state,
  including stale deletion at any depth.

Logs are appended to `~/.cache/hermes-context/sync.log` (real runs only),
always outside the mirrored tree.

## Safety guards

Refusals (nonzero exit, zero writes) cite the governing invariant:
- Degenerate paths (`/`, `.`, empty) refused.
- Source==destination by realpath, or ancestor/descendant containment
  (including symlinked ancestors resolving into the source) refused — the
  source always survives.
- Script-owned paths (log, staging, entrypoint) are validated against
  source/destination boundaries before anything is created.

## KNOWN_LIMITATIONS

- Exotic filenames (newlines/control characters) are out of scope for the
  losslessness guarantee.
- Adversarial environment-variable combinations beyond the built-in
  path-boundary guard are not defended.

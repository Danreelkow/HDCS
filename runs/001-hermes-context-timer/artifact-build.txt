=== sync-hermes-context.sh ===
#!/usr/bin/env bash
# sync-hermes-context.sh — host->workspace mirror of hermes context (A1, A2)
# Standalone-capable and systemd-invokable (A3). Guards cite A-numbers (A19).
set -u
set -o pipefail

# --- env: empty override is REFUSED, never falls back to default (ledger 8b) ---
get_env() { # $1=varname  $2=default
  local v="${!1-}"
  if [ -z "$v" ]; then
    echo "REFUSED A18: $1 is set but EMPTY; empty overrides are refused, no default fallback" >&2
    exit 1
  fi
  printf '%s' "$v"
}
if [ "${HERMES_CONTEXT_SRC+set}" = set ] || [ "${HERMES_CONTEXT_DST+set}" = set ]; then
  # only validate vars that are actually set-but-empty; unset vars take defaults
  if [ "${HERMES_CONTEXT_SRC+set}" = set ] && [ -z "${HERMES_CONTEXT_SRC}" ]; then
    echo "REFUSED A18: HERMES_CONTEXT_SRC is set but EMPTY" >&2; exit 1
  fi
  if [ "${HERMES_CONTEXT_DST+set}" = set ] && [ -z "${HERMES_CONTEXT_DST}" ]; then
    echo "REFUSED A18: HERMES_CONTEXT_DST is set but EMPTY" >&2; exit 1
  fi
fi
SRC="${HERMES_CONTEXT_SRC-/opt/data/workspace/hermes-context/}"
DST="${HERMES_CONTEXT_DST-/workspace/hermes-context/}"
LOG_DIR="${HERMES_CONTEXT_LOG_DIR-$HOME/.cache/hermes-context}"
LOG_FILE="$LOG_DIR/sync.log"

DRY_RUN=0
[ "${1-}" = "--dry-run" ] && DRY_RUN=1

die() { echo "REFUSED $1: $2" >&2; exit 1; }

# --- degenerate paths: '/', '', '.' via component test (A18) ---
degenerate_check() { # $1=path $2=name
  local p="$1"
  [ -z "$p" ] && die "A18" "$2 is empty"
  local norm="$p"
  while [ "${norm#/}" != "$norm" ]; do norm="${norm#/}"; done
  [ -z "$norm" ] && die "A18" "$2 resolves to '/'"
  [ "$norm" = "." ] && die "A18" "$2 is '.'"
}
degenerate_check "$SRC" "SRC"
degenerate_check "$DST" "DST"
degenerate_check "$LOG_DIR" "LOG_DIR"

# --- canonicalize once (ledger 7); all later mutation through canonical paths ---
SRC="${SRC%/}"
DST="${DST%/}"
LOG_DIR="${LOG_DIR%/}"

# --- component-boundary-aware containment (A14/A15), never string prefix ---
path_contains() { # $1=parent $2=child  -> 0 if parent == child or parent is a path component ancestor
  [ "$1" = "$2" ] && return 0
  case "$2/" in
    "$1"/*) return 0 ;;
  esac
  return 1
}

rp() { realpath -e -- "$1" 2>/dev/null; }

# --- A12 identity guard on both paths, pre-destruction ---
SRC_RP=$(rp "$SRC") || die "A12" "SRC ($SRC) does not exist or realpath failed"
DST_RP=$(rp "$DST" 2>/dev/null) || DST_RP=""
if [ -n "$DST_RP" ]; then
  path_contains "$SRC_RP" "$DST_RP" && die "A12" "DST is SRC, inside SRC, or SRC is inside DST"
  path_contains "$DST_RP" "$SRC_RP" && die "A12" "SRC is inside DST"
  # DST resolving through a symlink into SRC
  case "$DST_RP/" in
    "$SRC_RP"/*) die "A12" "DST resolves through symlink into SRC" ;;
  esac
  [ "$DST_RP" = "$SRC_RP" ] && die "A12" "realpath(SRC) == realpath(DST)"
fi

# --- A22: DST path-level symlink -> REFUSE, never replace ---
if [ -L "$DST" ]; then
  die "A22" "DST ($DST) is a symlink; operator must remove it manually"
fi

# --- A14: OWNED_PATH = concrete instantiated paths: stage parent, LOG_DIR, log file's PARENT dir, entrypoint (script) dir ---
SCRIPT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" >/dev/null 2>&1 && pwd -P) || SCRIPT_DIR=""
STAGE_PARENT="${TMPDIR:-/tmp}"
STAGE_PARENT="${STAGE_PARENT%/}"

owned_guard() { # $1=concrete owned path $2=label  (called only on existing paths)
  local o; o=$(rp "$1") || return 0
  [ -z "$o" ] && return 0
  [ "$o" = "/tmp" ] && return 0   # generic tmpdir is not script-owned
  if [ -n "$DST_RP" ]; then
    path_contains "$o" "$DST_RP" && die "A14" "DST is inside or equals OWNED_PATH ($2: $o)"
    path_contains "$DST_RP" "$o"  && die "A14" "DST contains OWNED_PATH ($2: $o)"
  fi
  # also DST-as-string against concrete owned path (DST may not exist yet)
  if path_contains "$o" "$DST" || path_contains "$DST" "$o"; then
    die "A14" "DST intersects OWNED_PATH ($2: $o)"
  fi
}

LOG_PARENT=$(dirname -- "$LOG_FILE")   # A14 concrete: log file's parent dir is owned
[ -e "$LOG_DIR" ] && owned_guard "$LOG_DIR" "LOG_DIR"
[ -e "$LOG_PARENT" ] && owned_guard "$LOG_PARENT" "log-file parent dir"
[ -n "$SCRIPT_DIR" ] && [ -e "$SCRIPT_DIR" ] && owned_guard "$SCRIPT_DIR" "entrypoint dir"
[ -e "$STAGE_PARENT" ] && owned_guard "$STAGE_PARENT" "stage parent"

# --- A20/A22: stage path computed as pure string, validated, THEN created via mktemp -d; re-validate instantiated path ---
make_stage() {
  local cand
  if [ -d "$STAGE_PARENT" ] && [ -w "$STAGE_PARENT" ]; then
    cand=$(mktemp -d "$STAGE_PARENT/hermes-stage.XXXXXX" 2>/dev/null) || cand=""
  else
    cand=""
  fi
  if [ -z "$cand" ]; then
    # script-owned fallback stage (A20)
    mkdir -p -- "$HOME/.cache/hermes-context-stage" || die "A20" "cannot create script-owned fallback stage"
    cand=$(mktemp -d "$HOME/.cache/hermes-context-stage/hermes-stage.XXXXXX") || die "A20" "mktemp failed"
  fi
  # re-validate the INSTANTIATED path (A22/A15)
  local inst; inst=$(rp "$cand") || { rm -rf -- "$cand"; die "A22" "instantiated stage path failed realpath"; }
  path_contains "$inst" "$DST" && { rm -rf -- "$cand"; die "A14" "instantiated stage contains DST"; }
  printf '%s' "$cand"
}

# --- MIRROR_CLASS compare: {file contents, recursive structure, symlinks} — NO dereference (A9, ledger 12) ---
compare_mirror() { # $1=a  $2=b  -> 0 iff equal on MIRROR_CLASS
  local a="$1" b="$2"
  # 1) regular contents + structure; --no-dereference: a regular file with the
  #    target's bytes must NOT pass for a symlink (lstat class, A9)
  diff -r --no-dereference "$a" "$b" >/dev/null 2>&1 || return 1
  # 2) symlink set must match exactly by relpath and link value (lstat-based)
  local la lb
  la=$( (cd -- "$a" && find . -mindepth 1 -type l -exec sh -c 'for f; do printf "%s %s\n" "${f#./}" "$(readlink -- "$f")"; done' _ {} + ) 2>/dev/null | LC_ALL=C sort)
  lb=$( (cd -- "$b" && find . -mindepth 1 -type l -exec sh -c 'for f; do printf "%s %s\n" "${f#./}" "$(readlink -- "$f")"; done' _ {} + ) 2>/dev/null | LC_ALL=C sort)
  [ "$la" = "$lb" ] || return 1
  return 0
}

if [ "$DRY_RUN" = 1 ]; then
  # ALL guards already ran; print plan only. ZERO writes: no log, no stage, no DST creation (A6, A16)
  echo "DRY-RUN plan: sync $SRC/ -> $DST/ (contents, recursive, --delete semantics; MIRROR_CLASS only)"
  if [ -d "$DST" ]; then
    if compare_mirror "$SRC" "$DST"; then
      echo "DRY-RUN: DST already mirrors SRC on MIRROR_CLASS; no changes needed"
    else
      echo "DRY-RUN: DST differs from SRC on MIRROR_CLASS; a real run would apply changes:"
      diff -r --no-dereference --brief "$SRC" "$DST" 2>/dev/null | head -50 || true
    fi
  else
    echo "DRY-RUN: DST ($DST) does not exist; a real run would create and populate it. Nothing created now (A16)."
  fi
  exit 0
fi

# --- real run: stage -> sync -> verify -> touch DST (order is law, A11) ---
STAGE=$(make_stage)
trap 'rm -rf -- "$STAGE"' EXIT

# sync SRC CONTENTS into stage, no nesting (A4)
if command -v rsync >/dev/null 2>&1; then
  rsync -a --delete "$SRC/" "$STAGE/" || { echo "ERROR: rsync stage fill failed" >&2; exit 1; }
else
  # cp -a fallback with explicit recursive stale-subtree deletion at every depth (A7)
  mkdir -p -- "$STAGE"
  cp -a "$SRC/." "$STAGE/" || { echo "ERROR: cp -a stage fill failed" >&2; exit 1; }
fi

# verify stage vs SRC on MIRROR_CLASS exactly (A9, A13)
if ! compare_mirror "$SRC" "$STAGE"; then
  echo "ERROR: stage verification FAILED vs SRC on MIRROR_CLASS; DST untouched (A9, A11, A13)" >&2
  exit 1
fi
# VERIFIED stage exists — destruction of DST is now lawful (A11)

mkdir -p -- "$DST" || { echo "ERROR: mkdir -p DST failed" >&2; exit 1; }
if command -v rsync >/dev/null 2>&1; then
  rsync -a --delete "$STAGE/" "$DST/" || { echo "ERROR: promotion rsync failed" >&2; exit 1; }
else
  rm -rf -- "$DST" && mkdir -p -- "$DST"
  cp -a "$STAGE/." "$DST/" || { echo "ERROR: promotion cp -a failed" >&2; exit 1; }
fi

# post-promotion idempotence check
compare_mirror "$SRC" "$DST" || { echo "ERROR: post-sync DST != SRC on MIRROR_CLASS" >&2; exit 1; }

# log one line, OUTSIDE DST (A8)
mkdir -p -- "$LOG_DIR"
echo "$(date -u +%Y-%m-%dT%H:%M:%SZ) synced $SRC -> $DST (MIRROR_CLASS, verified)" >> "$LOG_FILE"
echo "sync complete: $SRC -> $DST"
exit 0

=== README.md ===
# hermes-context sync

One-way mirror (host -> workspace) of the hermes context directory.

## Paths (env override, A17)

- `HERMES_CONTEXT_SRC` — default `/opt/data/workspace/hermes-context/`
- `HERMES_CONTEXT_DST` — default `/workspace/hermes-context/`
- `HERMES_CONTEXT_LOG_DIR` — default `$HOME/.cache/hermes-context`

An explicitly EMPTY env value (`HERMES_CONTEXT_SRC=""`, etc.) is REFUSED
(A18) — it never silently falls back to the default. Trailing slashes are
canonicalized once. All references above rename together via these three
variables.

The service unit's `ExecStart=%h/.config/systemd/user/sync-hermes-context.sh`
is the verbatim installed path; `%h` expands to the user's home.

## Usage

- Standalone: `~/.config/systemd/user/sync-hermes-context.sh [--dry-run]`
- systemd user units: `hermes-context.service` (Type=oneshot) + `hermes-context.timer`
  (`OnCalendar=*-*-* 00/6:00:00`, `Persistent=true`):
  `systemctl --user daemon-reload && systemctl --user enable --now hermes-context.timer`

## --dry-run: zero-write guarantee

`--dry-run` performs ALL guards, prints the plan/diff, and performs ZERO
writes of any kind: no log file, no stage directory, and if DST does not
exist it stays nonexistent (A6, A16).

## Sync semantics

- rsync (`-a --delete`) is primary; if rsync is absent, the `cp -a`
  fallback is used. The fallback deletes stale files AND stale subtrees at
  every depth before copying the verified stage, achieving the same exact
  mirror (A4, A5, A7).
- Sync is one-way host -> workspace (A2). Re-running on an already
  mirrored DST makes no changes and exits 0 (idempotent).
- Pipeline order is law: stage -> content-verify -> touch DST (A11, A13).

## MIRROR_CLASS scope

Verification enforces exactly {file contents, recursive directory
structure, symlinks}. Timestamps, permissions metadata nuances, and
hardlink identity are NOT part of the mirror class and are not preserved
or verified (A9). Verification never dereferences symlinks: a regular
file containing a symlink target's bytes does not pass for a symlink.

## Guards (refusals cite A-numbers, A19)

- Degenerate paths: SRC/DST empty, `.`, or `/` — refused (A18). Empty env
  override is refused, not defaulted (A18).
- Identity: realpath(SRC) == realpath(DST), either inside the other, or
  DST resolving through a symlink into SRC — refused (A12), checked
  pre-destruction on both paths.
- DST path-level symlink: REFUSED, never replaced — the operator removes
  the symlink manually (A22).
- Ownership: DST equal to, inside, or containing a concrete OWNED_PATH
  (stage dir, LOG_DIR, the log file's PARENT directory, entrypoint dir)
  is refused via component-boundary-aware comparison, never string
  prefix (A14, A15). Entrypoints and logs live outside the DST tree and
  the sync never deletes them (A8).

## KNOWN_LIMITATIONS

1. Exotic filenames (newline/control characters) break reporting and
   comparison; the corpus gate with `\n` in names is a known limitation,
   not a normal-operation failure (A21).
2. Adversarial env combinations beyond the A14 ownership guard are known
   limitations, closed by citation of A14 without re-derivation.

## Verification vocabulary

A copy is called "verified" only after it has been content-compared
against SRC on MIRROR_CLASS (A13). An unverified copy is never labeled
verified in this document or in script output.

=== sync-hermes-context.sh ===
#!/usr/bin/env bash
# sync-hermes-context.sh — one-way mirror $HERMES_CONTEXT_SRC -> /workspace/hermes-context/
# Source is parameterized via env HERMES_CONTEXT_SRC (required, namespaced; never bare SRC/DST).
# User-level; no root. Primary: rsync -a --delete. Fallback: verified-copy-then-swap.
set -u

SRC="${HERMES_CONTEXT_SRC:-}"
DST="/workspace/hermes-context/"
LOG_DIR="$HOME/.cache/hermes-context"
LOG_FILE="$LOG_DIR/sync.log"

# Source must come from the namespaced env var — never hardcoded, never bare SRC/DST.
if [ -z "$SRC" ]; then
    printf 'FAIL: env HERMES_CONTEXT_SRC is required (namespaced source path; never bare SRC/DST)\n' >&2
    exit 1
fi
# Trailing slash for contents-sync semantics (no nesting).
case "$SRC" in
    */) ;;
    *) SRC="$SRC/" ;;
esac

log() {
    # Never called in dry-run (dry-run skips log creation entirely).
    mkdir -p "$LOG_DIR"
    printf '%s %s\n' "$(date -Is)" "$*" >> "$LOG_FILE"
}

die() {
    printf 'FAIL: %s\n' "$*" >&2
    log "FAIL: $*" 2>/dev/null || true
    exit 1
}

# ---------- realpath identity guard (A12) ----------
# Runs pre-destructive, checks BOTH paths. On identity overlap: exit 0, zero writes.
guard_identity() {
    local rsrc rdst
    rsrc="$(realpath -e "$SRC" 2>/dev/null)" || die "cannot resolve SRC: $SRC"
    rdst="$(realpath -e "$DST" 2>/dev/null)" || die "cannot resolve DST: $DST"
    [ "$rsrc" = "$rdst" ] && return 1
    case "$rdst/" in
        "$rsrc"/*) return 1 ;;   # DST inside SRC (descendant)
    esac
    case "$rsrc/" in
        "$rdst"/*) return 1 ;;   # DST ancestor of SRC
    esac
    # symlink-into-SRC: any symlink component of DST resolving into SRC
    local p
    p="$(dirname "$rdst")"
    while [ "$p" != "/" ]; do
        if [ -L "$p" ]; then
            local rp
            rp="$(realpath -e "$p" 2>/dev/null)" || { p="$(dirname "$p")"; continue; }
            case "$rp" in
                "$rsrc"|"$rsrc"/*) return 1 ;;
            esac
        fi
        p="$(dirname "$p")"
    done
    return 0
}

DRY_RUN=0
[ "${1:-}" = "--dry-run" ] && DRY_RUN=1

# Guard runs in ALL modes (including dry-run) — zero writes either way.
if guard_identity; then
    :
else
    printf 'identity guard: SRC and DST overlap (equal/ancestor/descendant/symlink-into-SRC); no writes\n' >&2
    exit 0
fi

# ---------- dry-run (A6): read-only only, zero writes incl. logs ----------
if [ "$DRY_RUN" -eq 1 ]; then
    if command -v rsync >/dev/null 2>&1; then
        rsync -a --delete --dry-run --itemize-changes "$SRC" "$DST"
        rc=$?
    else
        # read-only diff-based listing; no writes anywhere
        diff -r --no-dereference "$SRC" "$DST" 2>&1 | sed 's/^/DRY-RUN would change: /'
        rc=$?
        [ "$rc" -eq 1 ] && rc=0   # differences reported; listing itself "succeeded"
    fi
    printf 'DRY-RUN complete: no writes performed (log not created)\n'
    exit "$rc"
fi

# ---------- primary path: rsync (A5) ----------
if command -v rsync >/dev/null 2>&1; then
    log "sync start (rsync) $SRC -> $DST"
    rsync -a --delete "$SRC" "$DST" || die "rsync failed"
else
    # ---------- fallback path (A4, A7, A11): verified copy FIRST, then destructive swap ----------
    log "sync start (fallback) $SRC -> $DST"
    TMP="$(mktemp -d "${TMPDIR:-/tmp}/hermes-sync.XXXXXX")" || die "mktemp failed"
    trap 'rm -rf "$TMP"' EXIT
    # copy CONTENTS of SRC (no nesting) into temp
    ( cd "$SRC" && tar -cf - . ) | ( cd "$TMP" && tar -xf - ) || die "tar-pipe copy failed"
    # verify the temp copy before any destruction of DST (A11)
    diff -r --no-dereference "$SRC" "$TMP" >/dev/null 2>&1 || die "temp copy verification failed; DST untouched"
    # now safe to remove stale DST and move verified copy into place
    rm -rf "$DST" || die "could not remove DST"
    mkdir -p "$(dirname "$DST")"
    mv "$TMP" "$DST" || die "move into place failed"
    trap - EXIT
fi

# ---------- post-sync verify (A9): contents + structure + symlinks ----------
verify() {
    local a="$1" b="$2" rel
    # Walk SRC; compare each entry's class and, for symlinks, target.
    while IFS= read -r -d '' rel; do
        local sa="$a$rel" sb="$b$rel"
        if [ -L "$sa" ] && [ -L "$sb" ]; then
            [ "$(readlink "$sa")" = "$(readlink "$sb")" ] || return 1
        elif [ -L "$sa" ] || [ -L "$sb" ]; then
            return 1
        elif [ -d "$sa" ] && [ -d "$sb" ]; then
            :
        elif [ -f "$sa" ] && [ -f "$sb" ]; then
            cmp -s "$sa" "$sb" || return 1
        else
            return 1
        fi
    done < <( cd "$a" && find . -mindepth 1 -print0 )
    # reverse walk: stale entries in DST are mismatches
    while IFS= read -r -d '' rel; do
        [ -e "$a$rel" ] || [ -L "$a$rel" ] || return 1
    done < <( cd "$b" && find . -mindepth 1 -print0 )
    return 0
}

if verify "$SRC" "$DST"; then
    log "sync OK: mirror equivalence verified"
    printf 'sync OK: %s -> %s (verified)\n' "$SRC" "$DST"
    exit 0
else
    die "post-sync verification FAILED: contents/structure/symlink mismatch"
fi

=== hermes-context.service ===
[Unit]
Description=Mirror hermes-context from host mount into workspace
After=local-fs.target

[Service]
Type=oneshot
# Source path is supplied via the namespaced env var HERMES_CONTEXT_SRC.
Environment=HERMES_CONTEXT_SRC=/opt/data/workspace/hermes-context/
ExecStart=%h/.local/bin/sync-hermes-context.sh

[Install]
WantedBy=default.target

Description=Run hermes-context sync every 6 hours

[Timer]
OnCalendar=*-*-* 0/6:00:00
Persistent=true
Unit=hermes-context.service

[Install]
WantedBy=timers.target

=== README.md ===
# hermes-context sync

One-way mirror of the canonical source tree into the workspace:
`/opt/data/workspace/hermes-context/` -> `/workspace/hermes-context/`

The source path is **parameterized**: the script reads the required, namespaced
env var `HERMES_CONTEXT_SRC` (never a bare `SRC`/`DST` name). It exits nonzero
if the variable is unset. The systemd unit supplies it via `Environment=`.

## must_keep

1. Source path: `/opt/data/workspace/hermes-context/` (canonical, never written to).
2. Dry-run zero-writes: `--dry-run` performs no writes at all, including no log
   file creation; DST is byte-identical before and after.
3. User-level systemd, no root: install as user units via
   `systemctl --user`; the script and units require no root privileges.

## Files

- `sync-hermes-context.sh` — the sync entrypoint. Install to `~/.local/bin/`
  (outside the mirrored tree, so sync never deletes its own entrypoint).
- `hermes-context.service` — user unit; `ExecStart=%h/.local/bin/sync-hermes-context.sh`,
  with `Environment=HERMES_CONTEXT_SRC=/opt/data/workspace/hermes-context/`.
- `hermes-context.timer` — user unit; runs every 6 hours (`OnCalendar=*-*-* 0/6:00:00`,
  `Persistent=true`).

## Install

```sh
mkdir -p ~/.local/bin ~/.config/systemd/user
install -m 0755 sync-hermes-context.sh ~/.local/bin/sync-hermes-context.sh
cp hermes-context.service hermes-context.timer ~/.config/systemd/user/
systemctl --user daemon-reload
systemctl --user enable --now hermes-context.timer
```

## Behavior

- Source: read from `HERMES_CONTEXT_SRC` (required; nonzero exit if unset).
- Primary path: `rsync -a --delete SRC/ DST/` (recursive mirror, stale entries deleted).
- Fallback (no rsync): SRC is copied to a temp dir, verified with `diff -r`,
  then DST is removed and the verified copy moved into place — recursive stale
  deletion at all depths, contents-sync with no nesting.
- Identity guard (realpath): if SRC and DST are equal, ancestor/descendant, or
  DST is a symlink into SRC, the script exits 0 with zero writes, checked on
  both paths before any destructive operation.
- Post-sync verification enforces mirror equivalence: contents, recursive
  directory structure, and symlink targets. Any mismatch -> nonzero exit with
  `FAIL` on stderr (never warn-exit-0).
- Log: `~/.cache/hermes-context/sync.log` — always outside DST; not created
  during dry-run.

=== rotate-hdcs-runs.sh ===
#!/usr/bin/env bash
# rotate-hdcs-runs.sh — rotate stale PATTERN files from RUNS_DIR into ARCHIVE_DIR.
# Default (no args) = dry-run: prints planned rotations/prunes, performs ZERO writes.
# --apply = sole writing mode. Toolchain: bash + coreutils only (no logrotate, no root).
set -u

SELF_DIR=$(cd "$(dirname "$(realpath -- "$0")")" && pwd) || exit 1
CONF="$SELF_DIR/hdcs-runs-rotation.conf"

conf_get() { sed -n "s/^$1=//p" "$CONF" | head -n 1; }

MODE=dry_run
if [ $# -gt 0 ]; then
  if [ "$1" = "--apply" ] && [ $# -eq 1 ]; then
    MODE=apply
  else
    echo "usage: $0 [--apply]" >&2
    exit 2
  fi
fi

refuse() { echo "REFUSED (A4): $1" >&2; exit 1; }

# --- A4 path law: env override allowed; set-but-empty refused; never a bare default word ---
runs=${HDCS_RUNS_DIR-$(conf_get RUNS_DIR)}
arch=${HDCS_ARCHIVE_DIR-$(conf_get ARCHIVE_DIR)}
age_days=$(conf_get AGE_DAYS)
keep=$(conf_get KEEP)
pattern=$(conf_get PATTERN)

[ -n "$runs" ] || refuse "HDCS_RUNS_DIR is set-but-empty"
[ -n "$arch" ] || refuse "HDCS_ARCHIVE_DIR is set-but-empty"

case "$runs" in ''|'.'|'/') refuse "degenerate RUNS_DIR";; esac
case "$arch" in ''|'.'|'/') refuse "degenerate ARCHIVE_DIR";; esac

c_runs=$(realpath -m -- "$runs") || refuse "RUNS_DIR cannot be canonicalized"
c_arch=$(realpath -m -- "$arch") || refuse "ARCHIVE_DIR cannot be canonicalized"

case "$c_runs" in '/'|'') refuse "degenerate RUNS_DIR after canonicalization";; esac
case "$c_arch" in '/'|'') refuse "degenerate ARCHIVE_DIR after canonicalization";; esac

# component-based containment (string-prefix is wrong for e.g. /tmp vs /tmp2)
inside() { # inside "$child" "$parent" -> 0 if child == parent or under it
  case "$1" in
    "$2"|"$2"/*) return 0 ;;
    *) return 1 ;;
  esac
}

[ "$c_runs" = "$c_arch" ] && refuse "RUNS_DIR and ARCHIVE_DIR are identical (path identity)"
inside "$c_arch" "$c_runs" && refuse "ARCHIVE_DIR is inside RUNS_DIR (containment)"
inside "$c_runs" "$c_arch" && refuse "RUNS_DIR is inside ARCHIVE_DIR (containment)"

case "$age_days" in ''|*[!0-9]*) echo "ERROR: bad AGE_DAYS in conf" >&2; exit 2;; esac
case "$keep" in ''|*[!0-9]*) echo "ERROR: bad KEEP in conf" >&2; exit 2;; esac
[ -n "$pattern" ] || { echo "ERROR: bad PATTERN in conf" >&2; exit 2; }

[ -d "$c_runs" ] || { echo "RUNS_DIR does not exist: $c_runs (nothing to do)" >&2; exit 0; }

now=$(date +%s)

# --- rotate stale files: stale <=> floor((now - mtime)/86400) >= AGE_DAYS (explicit epoch arithmetic, never bare -mtime +N) ---
while IFS= read -r -d '' f; do
  m=$(stat -c %Y -- "$f") || { echo "WARN: cannot stat $f, skipping" >&2; continue; }
  age=$(( (now - m) / 86400 ))
  [ "$age" -ge "$age_days" ] || continue
  rel=${f#"$c_runs"/}
  echo "rotate: $rel (age=${age}d >= ${age_days}d)"
  if [ "$MODE" = apply ]; then
    dest_dir="$c_arch/$(dirname -- "$rel")"
    mkdir -p -- "$dest_dir" || { echo "ERROR: mkdir failed for $dest_dir" >&2; exit 1; }
    base=$(basename -- "$rel")
    dest="$dest_dir/$base"
    s=1
    while [ -e "$dest" ]; do dest="$dest_dir/$base.$s"; s=$((s+1)); done
    if cp -p -- "$f" "$dest" && cmp -s -- "$f" "$dest"; then
      rm -f -- "$f" || { echo "ERROR: could not remove source $f" >&2; exit 1; }
    else
      echo "ERROR: lossless cmp failed for $rel; leaving source in place" >&2
      rm -f -- "$dest"
      exit 1
    fi
  fi
done < <(find "$c_runs" -type f -name "$pattern" -print0 | sort -z)

# --- prune ARCHIVE_DIR to KEEP newest; RUNS_DIR is never pruned ---
if [ -d "$c_arch" ]; then
  count=$(find "$c_arch" -type f | wc -l)
  if [ "$count" -gt "$keep" ]; then
    n=$((count - keep))
    find "$c_arch" -type f -printf '%T@ %p\n' | sort -n | head -n "$n" | cut -d' ' -f2- | while IFS= read -r p; do
      echo "prune: ${p#"$c_arch"/}"
      if [ "$MODE" = apply ]; then
        rm -f -- "$p"
      fi
    done
  fi
fi

exit 0
=== verify-rotation.sh ===
#!/usr/bin/env bash
# verify-rotation.sh — zero-write honesty checker. Exit 0 <=> conf parses (exactly the 5
# operator-fixed keys, verbatim values, no malformed lines) AND no stale file pending under
# RUNS_DIR AND archive listing intact. Fresh PATTERN matches do NOT fail.
# Every check feeds a flag accumulator (A7); no pipeline status is ever trusted bare.
set -u

SELF_DIR=$(cd "$(dirname "$(realpath -- "$0")")" && pwd) || exit 1
CONF="$SELF_DIR/hdcs-runs-rotation.conf"

flags=0
note() { echo "VERIFY-FAIL: $1" >&2; flags=1; }

# ---- check 1: conf parses — exactly 5 lines, each a mandated KEY=VALUE, verbatim ----
if [ ! -f "$CONF" ]; then
  note "conf missing: $CONF"
else
  n_lines=0
  n_runs=0; n_arch=0; n_age=0; n_pat=0; n_keep=0
  while IFS= read -r line || [ -n "$line" ]; do
    n_lines=$((n_lines + 1))
    if [ "$line" = "RUNS_DIR=/workspace/hdcs/runs" ]; then
      n_runs=$((n_runs + 1))
    elif [ "$line" = "ARCHIVE_DIR=/workspace/.hdcs-rotate/archive" ]; then
      n_arch=$((n_arch + 1))
    elif [ "$line" = "AGE_DAYS=14" ]; then
      n_age=$((n_age + 1))
    elif [ "$line" = "PATTERN=*.txt" ]; then
      n_pat=$((n_pat + 1))
    elif [ "$line" = "KEEP=50" ]; then
      n_keep=$((n_keep + 1))
    else
      note "conf line $n_lines does not parse as a mandated key/value: '$line'"
    fi
  done < "$CONF"
  [ "$n_lines" -eq 5 ] || note "conf must have exactly 5 lines, found $n_lines"
  [ "$n_runs" -eq 1 ] || note "RUNS_DIR key not present exactly once"
  [ "$n_arch" -eq 1 ] || note "ARCHIVE_DIR key not present exactly once"
  [ "$n_age" -eq 1 ] || note "AGE_DAYS key not present exactly once"
  [ "$n_pat" -eq 1 ] || note "PATTERN key not present exactly once"
  [ "$n_keep" -eq 1 ] || note "KEEP key not present exactly once"
fi

conf_get() { sed -n "s/^$1=//p" "$CONF" | head -n 1; }
age_days=$(conf_get AGE_DAYS)
pattern=$(conf_get PATTERN)
case "$age_days" in ''|*[!0-9]*) age_days=14;; esac
[ -n "$pattern" ] || pattern='*.txt'

# runtime dirs honor env overrides (A4: env override allowed)
runs=${HDCS_RUNS_DIR-/workspace/hdcs/runs}
arch=${HDCS_ARCHIVE_DIR-/workspace/.hdcs-rotate/archive}

# ---- check 2: no stale file pending under RUNS_DIR (explicit boundary, same law as rotate) ----
if [ -d "$runs" ]; then
  now=$(date +%s)
  while IFS= read -r -d '' f; do
    m=$(stat -c %Y -- "$f") || { note "cannot stat $f under RUNS_DIR"; continue; }
    age=$(( (now - m) / 86400 ))
    if [ "$age" -ge "$age_days" ]; then
      note "stale file pending rotation under RUNS_DIR: ${f#"$runs"/} (age=${age}d >= ${age_days}d)"
    fi
  done < <(find "$runs" -type f -name "$pattern" -print0)
fi

# ---- check 3: archive listing intact — names preserved with rel dirs; flat top-level entries are orphans ----
if [ -d "$arch" ]; then
  while IFS= read -r -d '' f; do
    rel=${f#"$arch"/}
    case "$rel" in
      */*) : ;;
      *) note "orphan/flat entry at ARCHIVE_DIR top level (rel dir lost): $rel" ;;
    esac
  done < <(find "$arch" -maxdepth 1 -type f -print0)
fi

# ---- verdict via flag accumulator only ----
if [ "$flags" -eq 0 ]; then
  exit 0
fi
exit 1
=== README.md ===
# hdcs rotation triad

Three files: `hdcs-runs-rotation.conf`, `rotate-hdcs-runs.sh`, `verify-rotation.sh`.

## Usage

```sh
./rotate-hdcs-runs.sh          # dry-run: prints planned rotations/prunes, ZERO writes
./rotate-hdcs-runs.sh --apply  # sole writing mode
./verify-rotation.sh           # zero-write honesty checker; exit 0 = healthy
```

Install: place both scripts and the conf in the same directory and `chmod +x` both scripts
(e.g. `~/.local/bin/` or the artifact dir — the conf is resolved relative to the script).

## Config keys (`hdcs-runs-rotation.conf`)

| Key         | Value                          | Meaning                                   |
|-------------|--------------------------------|-------------------------------------------|
| RUNS_DIR    | /workspace/hdcs/runs           | source tree scanned for stale files       |
| ARCHIVE_DIR | /workspace/.hdcs-rotate/archive| destination; mirrors RUNS_DIR subpaths    |
| AGE_DAYS    | 14                             | staleness threshold in whole days         |
| PATTERN     | *.txt                          | glob of rotated files                     |
| KEEP        | 50                             | archive pruned to this many newest files  |

Env overrides `HDCS_RUNS_DIR` / `HDCS_ARCHIVE_DIR` are honored; set-but-empty values are
refused (A4). Degenerate paths (`''`, `.`, `/`), identity, and containment in either
direction between RUNS_DIR and ARCHIVE_DIR are refused citing A4, before any write.

## Stale boundary (A5/A6)

A file is stale iff `floor((now - mtime) / 86400) >= AGE_DAYS`, computed with explicit
epoch arithmetic — never bare `find -mtime +N`. A file aged exactly AGE_DAYS days IS
rotated; one aged AGE_DAYS - 1 day 23h is NOT.

## Rotation and idempotence (A5)

`--apply` moves each stale file to `ARCHIVE_DIR/<relpath>` (relative path preserved),
appending a rotation suffix (`.1`, `.2`, ...) only on collision. The copy is `cmp`-verified
against the source before the source is removed — rotation is lossless. A second `--apply`
run is a byte-identical no-op (mtime is preserved on the archived copy, so it is never
re-selected). Pruning touches ARCHIVE_DIR only; RUNS_DIR is never pruned. Dry-run performs
zero writes — no archive dir is created, no state files are left.

## Verify semantics (A6/A7)

`verify-rotation.sh` exits 0 iff: the conf parses (exactly the 5 mandated keys, verbatim
values, no malformed lines), no stale file is pending under RUNS_DIR, and the archive
listing is intact (relative dirs preserved, no flat/orphan entries). A fresh PATTERN match
does not fail. Every check accumulates into a flag; no bare pipeline status decides the
exit. Verify is read-only.

## KNOWN_LIMITATIONS

- Exotic filenames (newlines, control characters): find is run NUL-delimited, but
  pathological names are not adversarially tested — treat as KNOWN_LIMITATIONS, not FAIL.
- Concurrent writers to RUNS_DIR during `--apply` are not locked; a file modified between
  the staleness check and the move may be archived at an older snapshot. Not a FAIL.
- `find -printf` is GNU findutils; on non-GNU systems the prune listing degrades.
